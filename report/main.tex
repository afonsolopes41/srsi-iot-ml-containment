\pendente{Contexto: crescimento da IoT, dispositivos com poucos recursos e sem
atualizações; exemplo da \textit{botnet} Mirai~\cite{antonakakis2017mirai}.}

\pendente{Problema: um dispositivo infetado continua na rede sem ser detetado; o
tráfego de rede é o único ponto de observação.}

\pendente{Ponto de partida: o trabalho de Noetzold \textit{et al.}~\cite{noetzold2026anomaly}
deteta anomalias e gera alertas, e aponta como trabalho futuro a resposta automática
a incidentes e a execução em \textit{edge}.}

\textbf{Objetivos.} \pendente{Responder às perguntas de pesquisa da Tabela~\ref{tab:perguntas}
e propor um sistema que deteta e isola automaticamente dispositivos IoT comprometidos.}

\textbf{Contribuições.} \pendente{(i) revisão sistemática de N artigos;
(ii) comparação do estado da arte (Tabela~\ref{tab:comparacao});
(iii) proposta de arquitetura para a Parte 2.}

A Secção~\ref{sec:metodologia} descreve o protocolo da revisão sistemática.

\subsection{\textit{Datasets}, tipos de ataque e métricas (QF2)}
% RESPONSÁVEL: Frente 2 (João Alves) (as equações das métricas são da Frente 3 (Rafael Silva))
\pendente{\textit{Datasets} mais usados (ex.: IoT-23~\cite{iot23}, CICIoT2023, N-BaIoT, TON\_IoT)
e porquê.}

As métricas mais usadas derivam da matriz de confusão, onde TP, FP, FN e TN são os
verdadeiros positivos, falsos positivos, falsos negativos e verdadeiros negativos:
\begin{equation}
\mathrm{Precision}=\frac{TP}{TP+FP},\qquad \mathrm{Recall}=\frac{TP}{TP+FN}
\label{eq:precrec}
\end{equation}
\begin{equation}
F_1 = \frac{2\cdot \mathrm{Precision}\cdot \mathrm{Recall}}{\mathrm{Precision}+\mathrm{Recall}}
\label{eq:f1}
\end{equation}
Num sistema que isola dispositivos, a taxa de falsos positivos mede o custo de
isolar um dispositivo legítimo:
\begin{equation}
FPR = \frac{FP}{FP+TN}
\label{eq:fpr}
\end{equation}

\subsection{Modelos de deteção}
% RESPONSÁVEL: Frentes 3 e 4
As \textit{features} numéricas de cada fluxo são normalizadas para $[0,1]$:
\begin{equation}
x' = \frac{x - x_{\min}}{x_{\max} - x_{\min}}
\label{eq:minmax}
\end{equation}

\pendente{Random Forest: uma frase.}

O \textit{Isolation Forest}~\cite{liu2008isolation} atribui a cada amostra $x$ um
\textit{score} de anomalia a partir do comprimento médio do caminho $h(x)$ necessário
para a isolar, numa amostra de $n$ observações:
\begin{equation}
s(x,n) = 2^{-\frac{E[h(x)]}{c(n)}},\qquad c(n) = 2H(n-1) - \frac{2(n-1)}{n}
\label{eq:iforest}
\end{equation}
onde $H(i)\approx \ln(i)+0{,}5772$. Valores de $s$ próximos de 1 indicam anomalia.

O \textit{Autoencoder} é treinado só com tráfego normal. Um fluxo com $d$ \textit{features}
é considerado anómalo quando o erro de reconstrução ultrapassa o limiar $\theta$:
\begin{equation}
e(x) = \frac{1}{d}\sum_{j=1}^{d}\left(x_j-\hat{x}_j\right)^2 > \theta
\label{eq:ae}
\end{equation}

\subsection{Avaliação planeada}
% RESPONSÁVEL: Frente 1 (Afonso)
Os três modelos serão comparados com as métricas (\ref{eq:precrec})--(\ref{eq:fpr}) e com a
\textit{Polygon Area Metric}~\cite{aydemir2020pam}, usada em~\cite{noetzold2026anomaly}, que
calcula a área do polígono formado por $n$ métricas normalizadas num gráfico radar:
\begin{equation}
PAM = \frac{1}{2}\left|\sum_{i=1}^{n}\left(x_i y_{i+1}-y_i x_{i+1}\right)\right|
\label{eq:pam}
\end{equation}
Será ainda medido o tempo até à contenção, desde o início do ataque até ao isolamento
do dispositivo:
\begin{equation}
T_c = t_{\text{isolamento}} - t_{\text{ataque}} = T_{\text{deteção}} + T_{\text{resposta}}
\label{eq:tc}
\end{equation}
